class zerosToEnd{
    public static void main(String[] args){
        int a=0;
        int arr[]={10,0,30,0,10,0,50,40};
        for(int i=0;i<arr.length;i++){
            for(int j=i+1;j<arr.length;j++){
                if(arr[i]==a){
                    int t=a;
                    arr[i]=arr[j];
                    arr[j]=t;
                }
            }
        }
        for(int i=0;i<arr.length;i++){
            System.out.print(arr[i]+" ");
        }
    }
}
