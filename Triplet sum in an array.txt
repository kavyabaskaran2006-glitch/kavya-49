class tripletSum{
    public static void main(String[] args){
        int t=10;
        int arr[]={2,3,5,4,7};
        for(int i=0;i<arr.length;i++){
            for(int j=i+1;j<arr.length;j++){
                for(int k=i+2;k<arr.length;k++){
                    if(arr[i]+arr[j]+arr[k]==t){
                    System.out.println(arr[i]+"+"+arr[j]+"+"+arr[k]+"="+t);
                }
            }
        }
    }
}
}
