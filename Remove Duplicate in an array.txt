class RemoveDuplicate{
    public static void main(String[] args){
        int a[]={10,20,10,30,40};
        for(int i=0;i<a.length;i++){
            boolean isduplicate=false;
            for(int j=0;j<i;j++){
                if(a[i]==a[j]){
                    isduplicate=true;
                    break;
                }
            }
        
        if(!isduplicate){
            System.out.println(a[i]);
        }
        }
    }
}
