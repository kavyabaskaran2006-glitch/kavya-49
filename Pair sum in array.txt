class PairSum{
    public static void main(String[] args){
        int t=7;
        int arr[]={2,4,3,5,7,8};
        for(int i=0;i<arr.length;i++){
            for(int j=i+1;j<arr.length;j++){
                if(arr[i]+arr[j]==t){
                    System.out.println(arr[i]+"+"+arr[j]+"="+t);
                }
            }
        }
    }
}
